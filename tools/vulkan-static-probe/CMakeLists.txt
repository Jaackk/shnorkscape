cmake_minimum_required(VERSION 3.26)
project(workspace_static_probe LANGUAGES CXX)
option(PROBE_FULL_WORKSPACE "V4 912-ID fixture capture" OFF)
option(PROBE_AUTO_CAPTURE "V5 disposable session-bound automatic Save capture" OFF)
if(PROBE_AUTO_CAPTURE AND NOT PROBE_FULL_WORKSPACE)
  message(FATAL_ERROR "Automatic capture requires the complete pinned workspace schema")
endif()
if(NOT DEFINED PROBE_IMAGE_HASH OR NOT DEFINED PROBE_THUNK_RVA)
  message(FATAL_ERROR "Use the generated manifest hash and thunk RVA")
endif()
add_library(shnork_workspace_probe SHARED probe.cpp)
target_compile_features(shnork_workspace_probe PRIVATE cxx_std_17)
target_compile_definitions(shnork_workspace_probe PRIVATE PROBE_IMAGE_HASH="${PROBE_IMAGE_HASH}" PROBE_THUNK_RVA=${PROBE_THUNK_RVA})
target_compile_options(shnork_workspace_probe PRIVATE /Brepro /W4)
target_link_options(shnork_workspace_probe PRIVATE /Brepro /INCREMENTAL:NO)
target_link_libraries(shnork_workspace_probe PRIVATE bcrypt)
set_target_properties(shnork_workspace_probe PROPERTIES OUTPUT_NAME shnork_workspace_probe_v3)
if(PROBE_FULL_WORKSPACE)
  target_compile_definitions(shnork_workspace_probe PRIVATE PROBE_FULL_WORKSPACE)
  set_target_properties(shnork_workspace_probe PROPERTIES OUTPUT_NAME shnork_workspace_probe_v4)
endif()
if(PROBE_AUTO_CAPTURE)
  target_compile_definitions(shnork_workspace_probe PRIVATE PROBE_AUTO_CAPTURE)
  target_link_libraries(shnork_workspace_probe PRIVATE iphlpapi ws2_32)
  set_target_properties(shnork_workspace_probe PROPERTIES OUTPUT_NAME shnork_workspace_probe_v5)
endif()
add_executable(probe_test probe.cpp)
target_compile_features(probe_test PRIVATE cxx_std_17)
target_compile_definitions(probe_test PRIVATE PROBE_TEST PROBE_IMAGE_HASH="${PROBE_IMAGE_HASH}" PROBE_THUNK_RVA=${PROBE_THUNK_RVA})
target_link_libraries(probe_test PRIVATE bcrypt)
if(PROBE_FULL_WORKSPACE)
  target_compile_definitions(probe_test PRIVATE PROBE_FULL_WORKSPACE)
endif()
enable_testing()
add_test(NAME bounded_lookup COMMAND probe_test)
if(PROBE_AUTO_CAPTURE)
  add_executable(auto_capture_test probe.cpp)
  target_compile_features(auto_capture_test PRIVATE cxx_std_17)
  target_compile_definitions(auto_capture_test PRIVATE PROBE_TEST PROBE_FULL_WORKSPACE PROBE_AUTO_CAPTURE PROBE_IMAGE_HASH="${PROBE_IMAGE_HASH}" PROBE_THUNK_RVA=${PROBE_THUNK_RVA})
  target_link_libraries(auto_capture_test PRIVATE bcrypt iphlpapi ws2_32)
  add_test(NAME automatic_capture_fence COMMAND auto_capture_test)
endif()
