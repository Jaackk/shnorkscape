cmake_minimum_required(VERSION 3.26)
project(vulkan_var_gate LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(FUNCHOOK_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FUNCHOOK_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(FUNCHOOK_INSTALL OFF CACHE BOOL "" FORCE)
if(NOT DEFINED FUNCHOOK_SOURCE)
  message(FATAL_ERROR "Supply the audited Funchook source path")
endif()
execute_process(COMMAND git -C "${FUNCHOOK_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE FH_REV OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT FH_REV STREQUAL "b4991704add411ecbc492dae020f375124d51f45")
  message(FATAL_ERROR "Unexpected Funchook revision")
endif()
add_subdirectory("${FUNCHOOK_SOURCE}" funchook)
add_library(var_gate SHARED gate.cpp)
target_include_directories(var_gate PRIVATE "${FUNCHOOK_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}/funchook")
target_link_libraries(var_gate PRIVATE funchook-static bcrypt)
target_compile_definitions(var_gate PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(var_gate_test gate.cpp)
target_include_directories(var_gate_test PRIVATE "${FUNCHOOK_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}/funchook")
target_link_libraries(var_gate_test PRIVATE funchook-static bcrypt)
target_compile_definitions(var_gate_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX GATE_TEST)
enable_testing()
add_test(NAME bounded_gate COMMAND var_gate_test)
